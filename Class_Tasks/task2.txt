# 1. Initialize a Git repository
git init

# 2. Add README.md and commit it
touch README.md
git add README.md
git commit -m "Add README file"

# 3. Modify a file and check changes before committing
nano README.md        # or use any editor
git status
git diff

# 4. Unstage a file after adding it
git add README.md
git reset README.md   # unstages the file

# 5. Fix a wrong commit message
git commit --amend -m "Correct commit message"

# 6. Revert a commit safely
git log               # copy the commit hash you want to revert
git revert <commit-hash>
