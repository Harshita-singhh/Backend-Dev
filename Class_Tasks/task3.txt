# 1. Create a new branch
git checkout -b feature-ui

# 2. Make changes and commit
nano app.txt          # edit the file
git add app.txt
git commit -m "Update UI text in feature branch"

# 3. Switch back to main branch
git checkout main

# 4. Make different changes in the same file (to simulate conflict)
nano app.txt
git add app.txt
git commit -m "Update UI text in main branch"

# 5. Merge feature-ui into main
git merge feature-ui
